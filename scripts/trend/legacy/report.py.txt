"""Publish the complete small-study matrix, including rejected hypotheses."""
import argparse
import csv
import datetime as dt
import hashlib
import json
from pathlib import Path
import statistics

ROOT = Path(__file__).resolve().parents[2]


def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def pct(value):
    return f"{value * 100:.2f}%" if value is not None else "—"


def num(value):
    return f"{value:.3f}" if value is not None else "—"


def verdict(study):
    choice = study["selection"]["id"]
    held = next(s for s in study["summaries"]["holdout"] if s["id"] == choice)
    stress = next(s for s in study["summaries"]["holdout"] if s["id"] == choice + "-stress")
    checks = {
        "developmentQualified": study["selection"]["developmentQualified"],
        "holdoutProfitableSymbols": held["profitableSymbols"] >= 4,
        "holdoutNetReturn": held["equalSleeveReturn"] > 0,
        "holdoutMeanNetR": (held["meanNetR"] or 0) > 0,
        "holdoutConfidence": held["dailyMean95CI"][0] > 0,
        "holdoutStress": stress["equalSleeveReturn"] > 0,
    }
    if "validation" in study["summaries"]:
        validation = next(s for s in study["summaries"]["validation"] if s["id"] == choice)
        checks["validation"] = validation["profitableSymbols"] >= 4 and validation["equalSleeveReturn"] > 0 and (validation["meanNetR"] or 0) > 0
    else:
        checks["holdoutSampleSize"] = all(s["metrics"]["trades"] >= 30 for s in held["symbols"])
    return {"accepted": all(checks.values()), "checks": checks}


def price_return(manifest, symbol, window):
    """Underlying price context only, deliberately not called a costed benchmark."""
    stamps = [int(dt.datetime.fromisoformat(window[k]).replace(tzinfo=dt.timezone.utc).timestamp() * 1000) for k in ["start", "end"]]
    prices = []
    for t, col in [(stamps[0], 1), (stamps[1] - 60000, 4)]:
        month = dt.datetime.fromtimestamp(t / 1000, dt.timezone.utc).strftime("%Y-%m")
        partition = next(p for p in manifest["symbols"][symbol]["partitions"] if p["month"] == month)
        with open(partition["candles"]) as f:
            row = next(r for r in csv.reader(f) if r[0] == str(t))
        prices.append(float(row[col]))
    return prices[1] / prices[0] - 1

            raise ValueError("plan hash mismatch; do not publish stale results")

            raise ValueError("development summary differs from the frozen choice")
        summaries = {w["id"]: read(directory / f"{w['id']}-summary.json") for w in plan["windows"]}
        study = {"id": index, "plan": str(plan_path.relative_to(ROOT)), "selection": selection,
                 "summaries": summaries, "provenance": [], "priceContext": {}}

                if receipt["resultSha256"] != sha(directory / window["id"] / f"{result['symbol']}.json") or result["planSha256"] != selection["planSha256"] or receipt["binarySha256"] != selection["binarySha256"]:
                    raise ValueError("raw outcome is not linked to the frozen selection")
            plot_candidates = [selection["id"]] + (["current-1m", "current-1h"] if index == 1 else ["channel-4h-both", "channel-1d-both"])
            for candidate in plot_candidates:
                candidate_rows = [next(r for r in b["results"] if r["id"] == candidate) for b in raw]
                curve = [{"time": p["time"], "nav": statistics.mean(r["daily"][i]["equity"] for r in candidate_rows) / plan["initialCapital"]} for i, p in enumerate(candidate_rows[0]["daily"])]
                curves.append({"study": index, "window": window["id"], "candidate": candidate, "points": curve})
            study["priceContext"][window["id"]] = {symbol: price_return(manifest, symbol, window) for symbol in plan["symbols"]}
            for b in raw:
                records = manifest["symbols"][b["symbol"]]["archives"]
                study["provenance"].append({"symbol": b["symbol"], "window": window["id"],
                    "resultSha256": sha(directory / window["id"] / f"{b['symbol']}.json"),
                    "fundingSha256": manifest["symbols"][b["symbol"]]["fundingSha256"],
                    "archives": [archive_ref(a) for a in records if any(month in Path(a["path"]).name for month in b["partitions"])]})
        study["verdict"] = verdict(study)
        studies.append(study)
    (args.output / "results.json").write_text(json.dumps({"version": 1, "engine": "trend-continuation-5", "studies": studies, "archives": archives}, indent=2) + "\n")
    (args.output / "curves.json").write_text(json.dumps(curves, separators=(",", ":")) + "\n")
    lines = ["# 简洁趋势策略：跨品种研究结果", "", "研究日期：2026-10-02。使用真实 Binance USDT 永续档案，非合成行情。", "",
        "![分阶段净值对照](equity.svg)", "",
        "结论：4 小时只做多通道方案值得继续模拟验证，观察到扣成本后的正期望；两轮研究均未满足预先声明的全部通过条件，不能称为已证实的长期正期望策略。", "",
        "实现候选：收盘突破此前 20 根高点，下一分钟开盘做多；入场冻结 ATR14，2 ATR 硬止损；收盘跌破此前 10 根低点，下分钟开盘退出。不开主动保本、不做分钟级追踪、不补仓，不预测行情。保留每笔 0.5% 风险、95% 名义敞口上限、连亏三次冷却与单日 3% 限制。", "",
        "## 研究方法", "",
        "- 品种：BTC、ETH、BNB、SOL、XRP、DOGE。每个独立账户初始 10,000 USDT；等额分仓收益是六账户期末资金的平均，无跨账户保证金或动态再平衡。",
